cmake_minimum_required(VERSION 3.15)
project(crp_bi_certifier_v4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
# Recent macOS SDKs ship libc++ headers inside the SDK.
if(APPLE AND EXISTS "${CMAKE_OSX_SYSROOT}/usr/include/c++/v1")
  include_directories(SYSTEM "${CMAKE_OSX_SYSROOT}/usr/include/c++/v1")
endif()
set(GC_EIGEN_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/vendor/eigen" CACHE PATH "Bundled Eigen")
set(SUITESPARSE OFF CACHE BOOL "Use bundled Eigen backend")
add_subdirectory(vendor/geometry-central)
add_executable(metric_probe metric_probe.cpp)
target_link_libraries(metric_probe geometry-central)
add_executable(gradient_check tests/gradient_check.cpp)
target_link_libraries(gradient_check geometry-central)
